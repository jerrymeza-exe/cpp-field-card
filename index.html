Valid Parentheses

Stack
stack<char> st;
unordered_map<char, char> match = {{')', '('}, {']', '['}, {'}', '{'}};
for (char c : s) {
    if (!match.count(c)) { st.push(c); continue; }   // opening bracket
    if (st.empty() || st.top() != match[c]) return false;
    st.pop();
}
return st.empty();

Sliding Window Maximum: monotonic deque

Deque
deque<int> dq;                 // indices; values decrease front to back
vector<int> out;
for (int i = 0; i < (int)nums.size(); i++) {
    while (!dq.empty() && nums[dq.back()] <= nums[i]) dq.pop_back();
    dq.push_back(i);
    if (dq.front() <= i - k) dq.pop_front();          // left the window
    if (i >= k - 1) out.push_back(nums[dq.front()]);
}
return out;

Grid BFS: fewest steps

Weeks 10–11
int rows = grid.size(), cols = grid[0].size();
vector<vector<bool>> seen(rows, vector<bool>(cols, false));
queue<pair<int, int>> q;
q.push({sr, sc});          // multi-source: push every start cell here
seen[sr][sc] = true;
int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
int steps = 0;
while (!q.empty()) {
    int levelSize = q.size();
    for (int i = 0; i < levelSize; i++) {
        auto [r, c] = q.front();
        q.pop();
        if (r == er && c == ec) return steps;
        for (auto& d : dirs) {
            int nr = r + d[0], nc = c + d[1];
            if (nr < 0 || nr >= rows || nc < 0 || nc >= cols) continue;
            if (seen[nr][nc] || grid[nr][nc] == 1) continue;   // 1 = wall
            seen[nr][nc] = true;   // mark on push, not on pop
            q.push({nr, nc});
        }
    }
    steps++;
}
return -1;

Grid DFS: Number of Islands

Week 10
int rows = grid.size(), cols = grid[0].size();
function<void(int, int)> dfs = [&](int r, int c) {
    if (r < 0 || r >= rows || c < 0 || c >= cols || grid[r][c] != '1') return;
    grid[r][c] = '0';                      // mark visited by sinking the land
    dfs(r + 1, c);
    dfs(r - 1, c);
    dfs(r, c + 1);
    dfs(r, c - 1);
};
int islands = 0;
for (int r = 0; r < rows; r++)
    for (int c = 0; c < cols; c++)
        if (grid[r][c] == '1') {
            islands++;
            dfs(r, c);
        }
return islands;

Union-find

Week 15
struct DSU {
    vector<int> parent, sz;
    DSU(int n) : parent(n), sz(n, 1) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];     // path halving
            x = parent[x];
        }
        return x;
    }
    bool unite(int a, int b) {             // false if already connected
        a = find(a);
        b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);     // smaller tree under larger
        parent[b] = a;
        sz[a] += sz[b];
        return true;
    }
};

Dijkstra

Week 16
vector<vector<pair<int, int>>> adj(n);     // adj[u] = {v, weight}
vector<int> dist(n, INT_MAX);
// {distance, node}, smallest distance on top
priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
dist[src] = 0;
pq.push({0, src});
while (!pq.empty()) {
    auto [d, u] = pq.top();
    pq.pop();
    if (d > dist[u]) continue;             // stale entry; already found shorter
    for (auto [v, w] : adj[u]) {
        if (d + w < dist[v]) {
            dist[v] = d + w;
            pq.push({dist[v], v});
        }
    }
}

Thread-safe queue

Week 32
template <typename T>
class SafeQueue {
public:
    void push(T item) {
        {
            lock_guard<mutex> lock(m_);
            q_.push(std::move(item));
        }
        cv_.notify_one();
    }

    T pop() {                              // blocks until an item arrives
        unique_lock<mutex> lock(m_);
        cv_.wait(lock, [this] { return !q_.empty(); });
        T item = std::move(q_.front());
        q_.pop();
        return item;
    }

private:
    queue<T> q_;
    mutex m_;
    condition_variable cv_;
};

Gotchas

Skim before each session

The mistakes that cost the most time. Most of them compile without complaint.

.size() is unsigned

v.size() - 1 on an empty vector is a huge number, not -1.

Fix(int)v.size() - 1

m[k] inserts

Checking with if (m[k]) adds k with value 0.

Fixm.count(k) to check

pop() returns nothing

int x = st.pop(); won't compile.

Fixint x = st.top(); st.pop();

Empty containers

top(), front() or back() on an empty container is undefined behavior.

FixCheck empty() first

int overflow

int tops out at 2,147,483,647.

Fixlong long, 1LL * a * b, l + (r - l) / 2

max needs matching types

max(best, v.size()) mixes int and an unsigned type.

Fixmax(best, (int)v.size())

Accidental copies

for (auto row : grid) copies each row. A vector<int> parameter copies on every recursive call.

Fixauto& and const vector<int>&

BFS visited timing

Marking visited when you pop lets one cell enter the queue many times.

FixMark when you push

Comparators must use <

<= in a sort comparator is undefined behavior and can crash.

Fixreturn a < b;

accumulate with 0

The sum is computed in int and can overflow.

Fixaccumulate(v.begin(), v.end(), 0LL)

No pair keys in unordered_map

There is no built-in hash for pair.

Fixmap<pair<int, int>, T> or r * cols + c

multiset::erase(x) removes all

Every copy of x disappears.

Fixms.erase(ms.find(x))

s = s + c in a loop

Copies the whole string each time, O(n²) overall.

Fixs += c

Deep recursion

DFS down a 100,000-node chain can overflow the stack.

FixAn explicit stack

Uninitialized variables

int sum; holds garbage, not 0.

Fixint sum = 0; and int cnt[26] = {};

bits/stdc++.h on a Mac

Apple Clang doesn't ship it.

FixInclude the headers you use

Data races

One thread writes a variable while another reads it with no lock: undefined behavior.

FixA mutex or atomic

Nothing matches. Try a shorter word, or press Esc to clear.
